create table public.families (
  id uuid not null default gen_random_uuid (),
  sabha_id uuid not null,
  family_code text null,
  address_multiline text null,
  payer_mobile text null,
  payer_email text null,
  opt_show_amount_in_directory boolean not null default false,
  opt_show_mobile_in_directory boolean not null default false,
  opt_show_email_in_directory boolean not null default false,
  created_by uuid null,
  created_at timestamp with time zone not null default now(),
  constraint families_pkey primary key (id),
  constraint families_family_code_key unique (family_code),
  constraint families_created_by_fkey foreign KEY (created_by) references auth.users (id),
  constraint families_sabha_id_fkey foreign KEY (sabha_id) references sabhas (id)
) TABLESPACE pg_default;

create table public.family_members (
  id uuid not null default gen_random_uuid (),
  family_id uuid not null,
  full_name text not null,
  age integer null,
  gender public.gender not null default 'U'::gender,
  gotra text null,
  amount numeric(12, 2) not null default 0,
  is_primary_payer boolean not null default false,
  created_at timestamp with time zone not null default now(),
  constraint family_members_pkey primary key (id),
  constraint family_members_family_id_fkey foreign KEY (family_id) references families (id) on delete CASCADE
) TABLESPACE pg_default;

create unique INDEX IF not exists one_primary_payer_per_family on public.family_members using btree (family_id) TABLESPACE pg_default
where
  (is_primary_payer = true);

create table public.profiles (
  user_id uuid not null,
  full_name text null,
  is_scm_office boolean not null default false,
  created_at timestamp with time zone not null default now(),
  constraint profiles_pkey primary key (user_id),
  constraint profiles_user_id_fkey foreign KEY (user_id) references auth.users (id) on delete CASCADE
) TABLESPACE pg_default;

create table public.receipt_email_dispatch (
  id uuid not null default gen_random_uuid (),
  entry_id uuid not null,
  receipt_no text not null,
  payer_email text null,
  status text not null default 'pending'::text,
  attempt_count integer not null default 0,
  last_error text null,
  provider_message_id text null,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone not null default now(),
  constraint receipt_email_dispatch_pkey primary key (id),
  constraint receipt_email_dispatch_entry_id_fkey foreign KEY (entry_id) references vantiga_entries (id) on delete CASCADE,
  constraint receipt_email_dispatch_attempt_count_check check ((attempt_count >= 0)),
  constraint receipt_email_dispatch_status_check check (
    (
      status = any (
        array[
          'pending'::text,
          'processing'::text,
          'sent'::text,
          'failed'::text
        ]
      )
    )
  )
) TABLESPACE pg_default;

create unique INDEX IF not exists receipt_email_dispatch_entry_receipt_uidx on public.receipt_email_dispatch using btree (entry_id, receipt_no) TABLESPACE pg_default;

create index IF not exists receipt_email_dispatch_status_updated_idx on public.receipt_email_dispatch using btree (status, updated_at) TABLESPACE pg_default;

create trigger trg_set_receipt_email_dispatch_updated_at BEFORE
update on receipt_email_dispatch for EACH row
execute FUNCTION set_receipt_email_dispatch_updated_at ();

create table public.sabha_remittances (
  id uuid not null default gen_random_uuid (),
  sabha_id uuid not null,
  fy text not null,
  remitted_at date not null,
  remitted_amount numeric(12, 2) not null,
  remittance_mode text not null,
  reference_no text null,
  status text not null default 'SUBMITTED'::text,
  created_by uuid not null,
  created_at timestamp with time zone null default now(),
  verified_by uuid null,
  verified_at timestamp with time zone null,
  rejection_reason text null,
  bank_name text null,
  remarks text null,
  constraint sabha_remittances_pkey primary key (id),
  constraint sabha_remittances_sabha_id_fkey foreign KEY (sabha_id) references sabhas (id),
  constraint sabha_remittances_verified_by_fkey foreign KEY (verified_by) references auth.users (id),
  constraint sabha_remittances_created_by_fkey foreign KEY (created_by) references auth.users (id),
  constraint sabha_remittances_remitted_amount_check check ((remitted_amount > (0)::numeric)),

        array[

          'NEFT/RTGS/IMPS'::text,
          'UPI'::text
        ]
      )
    )
  ),
  constraint sabha_remittances_status_check check (
    (
      status = any (
        array[
          'SUBMITTED'::text,
          'VERIFIED'::text,
          'REJECTED'::text
        ]
      )
    )
  )
) TABLESPACE pg_default;

create table public.vantiga_entries (
  id uuid not null default gen_random_uuid (),

    )
  );
